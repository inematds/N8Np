-- 1) Schema
CREATE SCHEMA IF NOT EXISTS nutri_bot;

-- 2) Funciones auxiliares para triggers
-- 2.1) set_updated_at(): actualiza el campo updated_at si existe en la fila
CREATE OR REPLACE FUNCTION nutri_bot.set_updated_at()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  IF TG_OP = 'UPDATE' THEN
    IF NEW.updated_at IS DISTINCT FROM now() THEN
      NEW.updated_at := now();
    END IF;
  END IF;
  RETURN NEW;
END;
$$;

-- 2.2) set_goal_start_metrics(): completa start_weight_kg / start_waist_cm
--     con el último check-in <= start_date si no fueron provistos.
CREATE OR REPLACE FUNCTION nutri_bot.set_goal_start_metrics()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  ref_date date;
  w_rec RECORD;
BEGIN
  ref_date := COALESCE(NEW.start_date, CURRENT_DATE);

  IF NEW.start_weight_kg IS NULL OR NEW.start_waist_cm IS NULL THEN
    SELECT w.weight_kg, w.waist_cm
      INTO w_rec
    FROM nutri_bot.weekly_checkins w
    WHERE w.patient_id = NEW.patient_id
      AND w.week_start_date <= ref_date
    ORDER BY w.week_start_date DESC
    LIMIT 1;

    IF NEW.start_weight_kg IS NULL AND w_rec.weight_kg IS NOT NULL THEN
      NEW.start_weight_kg := w_rec.weight_kg;
    END IF;

    IF NEW.start_waist_cm IS NULL AND w_rec.waist_cm IS NOT NULL THEN
      NEW.start_waist_cm := w_rec.waist_cm;
    END IF;
  END IF;

  -- Si la meta ya tiene end_date y status 'completed', mantenemos lo demás tal cual
  RETURN NEW;
END;
$$;

-- 3.1) patients
CREATE TABLE IF NOT EXISTS nutri_bot.patients (
  patient_id bigserial NOT NULL,
  full_name text NOT NULL,
  phone_e164 text NOT NULL,
  email text NULL,
  sex text NULL,
  height_cm numeric(5, 2) NULL,
  timezone text NOT NULL DEFAULT 'America/Bogota'::text,
  status text NOT NULL DEFAULT 'active'::text,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  birth_date date NULL,
  CONSTRAINT patients_patient_id_pkey PRIMARY KEY (patient_id),
  CONSTRAINT patients_phone_e164_key UNIQUE (phone_e164)
) TABLESPACE pg_default;

-- 3.2) weekly_checkins
CREATE TABLE IF NOT EXISTS nutri_bot.weekly_checkins (
  id bigserial NOT NULL,
  patient_id bigint NOT NULL,
  week_start_date date NOT NULL,
  weight_kg numeric(5, 2) NULL,
  waist_cm numeric(5, 2) NULL,
  energy_level smallint NULL,
  adherence smallint NULL,
  notes text NULL,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT weekly_checkins_pkey PRIMARY KEY (id),
  CONSTRAINT weekly_checkins_patient_id_week_start_date_key UNIQUE (patient_id, week_start_date),
  CONSTRAINT weekly_checkins_patient_id_fkey FOREIGN KEY (patient_id)
    REFERENCES nutri_bot.patients (patient_id) ON DELETE CASCADE
) TABLESPACE pg_default;

-- 3.3) goals
CREATE TABLE IF NOT EXISTS nutri_bot.goals (
  id bigserial NOT NULL,
  patient_id bigint NOT NULL,
  goal_type text NOT NULL,
  start_date date NOT NULL DEFAULT current_date,
  target_metric text NOT NULL,
  target_value numeric(7, 2) NOT NULL,
  deadline date NULL,
  status text NOT NULL DEFAULT 'active'::text,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  weekly_rate_kg numeric(4, 2) NULL,
  end_date date NULL,
  start_weight_kg numeric(6, 2) NULL,
  start_waist_cm numeric(6, 1) NULL,
  CONSTRAINT goals_pkey PRIMARY KEY (id),
  CONSTRAINT goals_patient_id_fkey FOREIGN KEY (patient_id)
    REFERENCES nutri_bot.patients (patient_id) ON DELETE CASCADE,
  CONSTRAINT goals_start_weight_chk CHECK (
    (start_weight_kg IS NULL) OR (start_weight_kg >= 20::numeric AND start_weight_kg <= 400::numeric)
  ),
  CONSTRAINT goals_status_chk CHECK (
    status = ANY (ARRAY['active'::text,'completed'::text,'cancelled'::text])
  ),
  CONSTRAINT goals_target_metric_chk CHECK (
    target_metric = ANY (ARRAY['weight_kg'::text,'body_fat_pct'::text,'bmi'::text,'waist_cm'::text])
  ),
  CONSTRAINT goals_start_waist_chk CHECK (
    (start_waist_cm IS NULL) OR (start_waist_cm >= 30::numeric AND start_waist_cm <= 200::numeric)
  ),
  CONSTRAINT goals_goal_type_chk CHECK (
    goal_type = ANY (ARRAY['lose'::text,'maintain'::text,'gain'::text])
  )
) TABLESPACE pg_default;

-- 5.1) v_active_goal
CREATE OR REPLACE VIEW nutri_bot.v_active_goal AS
SELECT
  g.id,
  g.patient_id,
  g.goal_type,
  g.start_date,
  g.target_metric,
  g.target_value,
  g.deadline,
  g.status,
  g.created_at,
  g.updated_at,
  g.weekly_rate_kg,
  g.end_date
FROM nutri_bot.goals g
WHERE g.status = 'active'::text;

-- 5.2) v_patient_latest_weight
CREATE OR REPLACE VIEW nutri_bot.v_patient_latest_weight AS
SELECT DISTINCT ON (w.patient_id)
  w.patient_id,
  w.week_start_date,
  w.weight_kg,
  w.created_at
FROM nutri_bot.weekly_checkins w
WHERE w.weight_kg IS NOT NULL
ORDER BY w.patient_id, w.week_start_date DESC;

-- 5.3) v_weight_time_series
CREATE OR REPLACE VIEW nutri_bot.v_weight_time_series AS
SELECT
  p.patient_id,
  p.full_name,
  w.week_start_date,
  w.weight_kg
FROM nutri_bot.patients p
JOIN nutri_bot.weekly_checkins w ON w.patient_id = p.patient_id
WHERE w.weight_kg IS NOT NULL
ORDER BY p.patient_id, w.week_start_date;

-- 5.4) v_patient_overview
CREATE OR REPLACE VIEW nutri_bot.v_patient_overview AS
WITH last_weight AS (
  SELECT
    w.patient_id,
    w.weight_kg,
    w.week_start_date,
    ROW_NUMBER() OVER (PARTITION BY w.patient_id ORDER BY w.week_start_date DESC) AS rn
  FROM nutri_bot.weekly_checkins w
  WHERE w.weight_kg IS NOT NULL
)
SELECT
  p.patient_id,
  p.full_name,
  p.phone_e164,
  p.email,
  p.sex,
  p.birth_date,
  p.height_cm,
  p.timezone,
  p.status,
  DATE_PART('year', age(p.birth_date::timestamp with time zone))::integer AS age_years,
  pref.daily_contact_hour,
  pref.portion_style,
  prof.activity_level,
  prof.dietary_pattern,
  prof.allergies,
  prof.dislikes,
  prof.meals_per_day,
  prof.sleep_window,
  prof.equipment,
  g.id AS goal_id,
  g.goal_type,
  g.target_metric,
  g.target_value,
  g.weekly_rate_kg,
  g.deadline,
  g.status AS goal_status,
  lw.weight_kg AS latest_weight_kg,
  lw.week_start_date AS latest_weight_week,
  CASE
    WHEN p.height_cm IS NOT NULL AND p.height_cm > 0::numeric AND lw.weight_kg IS NOT NULL
      THEN ROUND(lw.weight_kg / ((p.height_cm / 100.0) ^ 2::numeric), 2)
    ELSE NULL::numeric
  END AS bmi
FROM nutri_bot.patients p
LEFT JOIN nutri_bot.patient_preferences pref ON pref.patient_id = p.patient_id
LEFT JOIN nutri_bot.patient_profile    prof ON prof.patient_id = p.patient_id
LEFT JOIN nutri_bot.v_active_goal      g    ON g.patient_id = p.patient_id
LEFT JOIN last_weight                  lw   ON lw.patient_id = p.patient_id AND lw.rn = 1;

-- 6) Triggers (creación condicional)
-- ============================================

-- goals.updated_at
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_trigger WHERE tgname = 'trg_goals_updated_at'
  ) THEN
    CREATE TRIGGER trg_goals_updated_at
    BEFORE UPDATE ON nutri_bot.goals
    FOR EACH ROW
    EXECUTE FUNCTION nutri_bot.set_updated_at();
  END IF;
END$$;

-- goals.set_start_metrics
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_trigger WHERE tgname = 'trg_goals_set_start_metrics'
  ) THEN
    CREATE TRIGGER trg_goals_set_start_metrics
    BEFORE INSERT OR UPDATE ON nutri_bot.goals
    FOR EACH ROW
    EXECUTE FUNCTION nutri_bot.set_goal_start_metrics();
  END IF;
END$$;
