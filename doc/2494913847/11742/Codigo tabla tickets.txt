create table
  public.tickets_soporte (
    id serial not null,
    nombre character varying(100) null,
    email character varying(100) null,
    mensaje text null,
    sentimiento character varying(20) null,
    urgencia character varying(20) null,
    categoria character varying(50) null,
    timestamp timestamp with time zone null default now(),
    resuelto boolean null default false,
    constraint tickets_soporte_pkey primary key (id)
  ) tablespace pg_default;
